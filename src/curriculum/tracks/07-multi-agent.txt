=== ma-01 | The supervisor pattern | 4
::: story
One LLM-powered SUPERVISOR decides which specialist WORKER acts next; each
worker reports back to the supervisor; the supervisor says FINISH when done.
```
              +------------+
  START ----> | supervisor | --FINISH--> END
              +------------+
               ^   |   ^   |
               |   v   |   v
            researcher  coder
```
::: setup fake_llm mini_graph
::: task
Write `build_supervisor(llm, workers, max_turns=4)` where `workers` is a dict
name -> function `fn(state) -> str`. State `TeamState`: `task: str`,
`messages: Annotated[list, operator.add]`, `next: str`, `turns: int`.
- `supervisor` node: if `turns` (missing = 0) >= max_turns, next = "FINISH".
  Otherwise ask the llm with this exact prompt:
      Task: <task>
      Workers: <names joined by ", ">
      History:
      <one line per message: "<name>: <content>">
      Who acts next? Reply with a worker name or FINISH.
  Strip the reply; anything that isn't a worker name becomes "FINISH".
- one node per worker: returns
  `{"messages": [{"name": <worker>, "content": fn(state)}], "turns": turns + 1}`
- edges: START -> supervisor; supervisor -> (next) -> worker or END;
  every worker -> supervisor.
Beware Python's late-binding closures when creating nodes in a loop!

import operator
from typing import Annotated, TypedDict

class TeamState(TypedDict):
    task: str
    messages: Annotated[list, operator.add]
    next: str
    turns: int

def build_supervisor(llm, workers, max_turns=4):
    names = list(workers)

    def supervisor(state):
        if state.get("turns", 0) >= max_turns:
            return {"next": "FINISH"}
        history = "\n".join(f"{m['name']}: {m['content']}" for m in state.get("messages", []))
        prompt = (
            f"Task: {state['task']}\n"
            f"Workers: {', '.join(names)}\n"
            f"History:\n{history}\n"
            "Who acts next? Reply with a worker name or FINISH."
        )
        choice = llm.invoke(prompt).strip()
        return {"next": choice if choice in workers else "FINISH"}

    def make_worker(name, fn):
        def node(state):
            return {"messages": [{"name": name, "content": fn(state)}], "turns": state.get("turns", 0) + 1}
        return node

    graph = StateGraph(TeamState)
    graph.add_node("supervisor", supervisor)
    for name, fn in workers.items():
        graph.add_node(name, make_worker(name, fn))
        graph.add_edge(name, "supervisor")
    graph.add_edge(START, "supervisor")
    graph.add_conditional_edges("supervisor", lambda s: s["next"], {**{n: n for n in names}, "FINISH": END})
    return graph.compile()
::: hint
A helper `make_worker(name, fn)` that returns a node function captures the right name for each worker. A lambda inside the loop would see only the LAST name.
::: hint
Path map: `{**{n: n for n in names}, "FINISH": END}`.

=== ma-02 | Handoffs with Command | 4
::: story
Instead of a central router, agents can HAND OFF directly. A node returns a
`Command` that both updates state and names the next node:
```
return Command(update={"replies": ["..."]}, goto="billing")
```
No conditional edges needed — the graph only needs an entry edge.

import operator
from typing import Annotated, TypedDict

class DeskState(TypedDict):
    request: str
    replies: Annotated[list, operator.add]
    handoffs: Annotated[list, operator.add]

def triage(state):
    text = state["request"].lower()
    if "refund" in text or "invoice" in text:
        return Command(update={"handoffs": ["triage->billing"]}, goto="billing")
    if "error" in text or "crash" in text:
        return Command(update={"handoffs": ["triage->tech"]}, goto="tech")
    return Command(update={"replies": ["triage: how can I help?"]}, goto=END)

def tech(state):
    update = {"replies": ["tech: restart the app"]}
    if "refund" in state["request"].lower():
        return Command(update={**update, "handoffs": ["tech->billing"]}, goto="billing")
    return Command(update=update, goto=END)

def billing(state):
    return Command(update={"replies": ["billing: refund issued"]}, goto=END)

def build_helpdesk():
    graph = StateGraph(DeskState)
    graph.add_node("triage", triage)
    graph.add_node("tech", tech)
    graph.add_node("billing", billing)
    graph.add_edge(START, "triage")
    return graph.compile()
::: hint
`Command(update={...}, goto="billing")` — `goto=END` finishes the run.
::: hint
Since the reducers are `operator.add`, updates must be LISTS: `{"handoffs": ["triage->tech"]}`.

import operator
from typing import Annotated, TypedDict

class ResearchState(TypedDict):
    question: str
    notes: Annotated[list, operator.add]
    summary: str

class TeamState(TypedDict):
    question: str
    messages: Annotated[list, operator.add]

def build_researcher():
    def search_node(state):
        return {"notes": search(state["question"])}

    def summarize_node(state):
        return {"summary": "; ".join(state["notes"])}

    g = StateGraph(ResearchState)
    g.add_node("search_node", search_node)
    g.add_node("summarize_node", summarize_node)
    g.add_edge(START, "search_node")
    g.add_edge("search_node", "summarize_node")
    g.add_edge("summarize_node", END)
    return g.compile()

def build_team():
    researcher_graph = build_researcher()

    def researcher(state):
        result = researcher_graph.invoke({"question": state["question"]})
        return {"messages": [{"name": "researcher", "content": result["summary"]}]}

    def writer(state):
        research = next(m["content"] for m in state["messages"] if m["name"] == "researcher")
        return {"messages": [{"name": "writer", "content": "Report: " + research}]}

    g = StateGraph(TeamState)
    g.add_node("researcher", researcher)
    g.add_node("writer", writer)
    g.add_edge(START, "researcher")
    g.add_edge("researcher", "writer")
    g.add_edge("writer", END)
    return g.compile()
::: hint
The parent node returns ONLY the summary — the subgraph's notes never leak into the parent state.
::: hint

=== ma-04 | Planner, executor, critic | 5
::: story
A robust agent team for multi-step goals:
- PLANNER asks the LLM for a numbered plan
- EXECUTOR runs the steps one at a time (looping on itself)
- CRITIC checks the results; on failure it can send the team back to the
  planner (with the failures as feedback) — but only a limited number of times
`run_step(step)` is provided: it returns `"did <step>"`, or raises if the
step contains "impossible".
::: setup fake_llm mini_graph
def run_step(step):
    if "impossible" in step.lower():
        raise RuntimeError(f"cannot {step}")
    return f"did {step}"
::: task
1. `parse_plan(text)` -> list of steps from lines like `"1. foo"`, `"2) bar"`,
   `"- baz"` (strip the numbering/bullets; skip blank lines).
2. `build_pec(llm, max_replans=1)` -> compiled graph over `PECState`
   (`goal`, `plan`, `step`, `results`, `verdict`, `replans`), all plain
   (overwrite) keys:
   - `planner`: prompt `"Plan steps for: <goal>"`, plus
     `"\nPrevious attempt failed: <failed results joined by '; '>"` if any
     current results start with "FAILED". Returns the parsed plan, `step=0`,

   - `executor`: runs `run_step(plan[step])`; an exception becomes the result
     `"FAILED: <message>"`. Appends to results, increments step.

import re
from typing import TypedDict

class PECState(TypedDict):
    goal: str
    plan: list
    step: int
    results: list
    verdict: str
    replans: int

def parse_plan(text):
    steps = []
    for line in text.splitlines():
        line = re.sub(r"^\s*(?:\d+[.)]|[-*])\s*", "", line).strip()
        if line:
            steps.append(line)
    return steps

def _failed(state):
    return [r for r in state.get("results", []) if r.startswith("FAILED")]

def build_pec(llm, max_replans=1):
    def planner(state):
        prompt = f"Plan steps for: {state['goal']}"
        failed = _failed(state)
        if failed:
            prompt += f"\nPrevious attempt failed: {'; '.join(failed)}"
        replans = state.get("replans", 0) + (1 if "plan" in state else 0)
        return {"plan": parse_plan(llm.invoke(prompt)), "step": 0, "results": [], "replans": replans}

    def executor(state):
        step_text = state["plan"][state["step"]]
        try:
            result = run_step(step_text)
        except Exception as exc:
            result = f"FAILED: {exc}"
        return {"results": state["results"] + [result], "step": state["step"] + 1}

    def critic(state):
        failed = _failed(state)
        if failed and state["replans"] < max_replans:
            return {"verdict": "replan"}
        return {"verdict": "give_up" if failed else "accept"}

    g = StateGraph(PECState)
    g.add_node("planner", planner)
    g.add_node("executor", executor)
    g.add_node("critic", critic)
    g.add_edge(START, "planner")
    g.add_edge("planner", "executor")
    g.add_conditional_edges("executor", lambda s: "executor" if s["step"] < len(s["plan"]) else "critic")
    g.add_conditional_edges("critic", lambda s: "planner" if s["verdict"] == "replan" else END)
    return g.compile()
::: hint
`re.sub(r"^\s*(?:\d+[.)]|[-*])\s*", "", line)` strips "1." "2)" "-" "*".
::: hint
A router can return node names directly without a path map: `lambda s: "executor" if s["step"] < len(s["plan"]) else "critic"`.

import operator
from typing import Annotated, TypedDict

class MRState(TypedDict):
    topics: list
    summaries: Annotated[list, operator.add]
    final: str

def build_map_reduce(llm):
    def fan_out(state):
        topics = state.get("topics", [])
        if not topics:
            return "reduce"
        return [Send("summarize", {"topic": t}) for t in topics]

    def summarize(arg):
        text = llm.invoke("Summarize " + arg["topic"])
        return {"summaries": [f"{arg['topic']}: {text}"]}

    def reduce(state):
        items = sorted(state.get("summaries", []))
        return {"final": "\n".join(items) if items else "(nothing to summarize)"}

    g = StateGraph(MRState)
    g.add_node("summarize", summarize)
    g.add_node("reduce", reduce)
    g.add_conditional_edges(START, fan_out)
    g.add_edge("summarize", "reduce")
    g.add_edge("reduce", END)
    return g.compile()
::: hint

::: hint

import operator
from typing import Annotated, TypedDict

class ResearchState(TypedDict):
    topic: str
    sources: list
    draft: str
    feedback: str
    revisions: int
    reviewed: bool
    approved: bool
    next: str
    log: Annotated[list, operator.add]

def run_research(llm, topic, max_revisions=2):
    def supervisor(state):
        if not state.get("sources"):
            nxt = "researcher"
        elif not state.get("draft"):
            nxt = "writer"
        elif not state.get("reviewed"):
            nxt = "reviewer"
        elif state.get("approved"):
            nxt = "FINISH"
        elif state.get("revisions", 0) < max_revisions:
            nxt = "writer"
        else:
            nxt = "FINISH"
        return {"next": nxt}

    def researcher(state):
        return {"sources": search(state["topic"]), "log": ["researcher"]}

    def writer(state):
        titles = ", ".join(s["title"] for s in state["sources"])
        prompt = f"Write about {state['topic']} using: {titles}"
        if state.get("feedback"):
            prompt += f"\nAddress feedback: {state['feedback']}"
        body = llm.invoke(prompt)
        refs = "\n".join(f"- {s['title']} ({s['url']})" for s in state["sources"])
        return {
            "draft": f"{body}\n\nSources:\n{refs}",
            "revisions": state.get("revisions", 0) + 1,
            "reviewed": False,
            "log": ["writer"],
        }

    def reviewer(state):
        reply = llm.invoke("Review this draft:\n" + state["draft"])
        update = {"reviewed": True, "log": ["reviewer"]}
        if reply.startswith("APPROVE"):
            update["approved"] = True
        elif reply.startswith("REVISE:"):
            update["feedback"] = reply.split(":", 1)[1].strip()
        return update

    g = StateGraph(ResearchState)
    g.add_node("supervisor", supervisor)
    for name, fn in [("researcher", researcher), ("writer", writer), ("reviewer", reviewer)]:
        g.add_node(name, fn)
        g.add_edge(name, "supervisor")
    g.add_edge(START, "supervisor")
    g.add_conditional_edges("supervisor", lambda s: s["next"],
                            {"researcher": "researcher", "writer": "writer", "reviewer": "reviewer", "FINISH": END})
    out = g.compile().invoke({"topic": topic})
    return {
        "report": out.get("draft", ""),
        "approved": bool(out.get("approved")),
        "revisions": out.get("revisions", 0),
        "log": out.get("log", []),
    }
::: hint
